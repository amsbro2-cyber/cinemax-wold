import { useCallback, useEffect, useState } from 'react'
import { useHashRoute } from '@/hooks/useHashRoute'
import { MOVIES } from '@/data/movies'
import Navbar from '@/components/Navbar'
import Footer from '@/components/Footer'
import ApiKeyDialog from '@/components/ApiKeyDialog'
import Home from '@/pages/Home'
import MovieDetailsPage from '@/pages/MovieDetailsPage'

export default function App() {
  const { route, goHome, openMovie } = useHashRoute()
  const [query, setQuery] = useState('')
  const [category, setCategory] = useState('All')
  const [keyDialogOpen, setKeyDialogOpen] = useState(false)
  const [apiKeyVersion, setApiKeyVersion] = useState(0)

  useEffect(() => {
    document.title =
      route.name === 'movie'
        ? 'Movie Details — CINEMAX WORLD'
        : 'CINEMAX WORLD — Discover Movies'
  }, [route])

  const focusSearch = useCallback(() => {
    goHome()
    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        const el = document.querySelector<HTMLInputElement>('#hero-search')
        el?.focus()
        el?.scrollIntoView({ behavior: 'smooth', block: 'center' })
      })
    })
  }, [goHome])

  const pickCategory = useCallback(
    (c: string) => {
      setCategory(c)
      goHome()
      requestAnimationFrame(() => {
        requestAnimationFrame(() => {
          document.querySelector('#browse')?.scrollIntoView({ behavior: 'smooth', block: 'start' })
        })
      })
    },
    [goHome],
  )

  return (
    <div className="min-h-dvh bg-[hsl(var(--bg))] text-[hsl(var(--fg))]">
      <Navbar onGoHome={goHome} onSearchFocus={focusSearch} onOpenKeyDialog={() => setKeyDialogOpen(true)} />

      {route.name === 'home' ? (
        <Home
          movies={MOVIES}
          query={query}
          onQueryChange={setQuery}
          category={category}
          onCategoryChange={setCategory}
          onOpen={openMovie}
          apiKeyVersion={apiKeyVersion}
        />
      ) : (
        <MovieDetailsPage
          id={route.id}
          movies={MOVIES}
          onBack={goHome}
          onOpen={openMovie}
          apiKeyVersion={apiKeyVersion}
        />
      )}

      <Footer onCategory={pickCategory} />

      <ApiKeyDialog
        open={keyDialogOpen}
        onClose={() => setKeyDialogOpen(false)}
        onSaved={() => setApiKeyVersion((v) => v + 1)}
      />
    </div>
  )
}
